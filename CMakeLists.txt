# CMake build of GMIC.ofx.bundle, the only build system (Natron's build-plugins.sh and CI use it):
#
#   cmake -S . -B build -G Ninja -DCMAKE_BUILD_TYPE=Release   # Debug, RelWithDebInfo
#   cmake --build build                    # assembles build/Plugins/GMIC.ofx.bundle
#   cmake --install build --prefix <dir>   # installs <dir>/OFX/Plugins/GMIC.ofx.bundle
#
# Options: -DOFX_CXX_STANDARD=20 (default: CMAKE_CXX_STANDARD, or 17), -DOFX_USE_OPENMP=OFF
# (default ON), -DOFX_LIBOMP_ROOT=<dir> (Apple clang: libomp, default MacPorts'),
# -DOFX_DEPLOY_BUNDLES=ON (macOS: the install step copies the MacPorts libraries into the bundle,
# for release builds), -DGMIC_TARBALL=<path> (a downloaded gmic_<version>.tar.gz, for offline
# builds), -DGMIC_POCKETFFT=<path> (a downloaded pocketfft_hdronly.h, likewise), -DGMICVER=...
# -DGMICSRC_SHA256=... (see gmic_sources.txt), -DOFX_LICENSE=COMMERCIAL (Natron's NATRON_LICENSE)
# and -DFFT=fftw3 (default: pocketfft).

# CMake 3.21, as openfx's modules need; the policies up to 3.31 are NEW (later ones stay OLD: raise
# the upper bound after testing a newer CMake)

# ========================== G'MIC sources ==========================
#
# Downloaded and verified, and extracted into the build tree (never into the source directory). The
# version, checksums, pocketfft pin and local patches are read from gmic_sources.txt, the single
# place to change when upgrading G'MIC (GMICVER, GMICSRC_SHA256_<version>). -DGMICVER=<version>
# -DGMICSRC_SHA256=<sha256> override them; since CMake keeps them in its cache, a warning tells when
# they differ from gmic_sources.txt (e.g. after a GMICVER bump): remove them with -UGMICVER
# -UGMICSRC_SHA256.

set(GMIC_SOURCES_FILE "${PROJECT_SOURCE_DIR}/gmic_sources.txt")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${GMIC_SOURCES_FILE}")
file(STRINGS "${GMIC_SOURCES_FILE}" _gmic_pins)
# Only "NAME = value" lines starting in the first column (as the other readers of the file match
# them, see its top), each name once, and comment and blank lines: fail on anything else, rather
# than read a wrong value or ignore one.
set(_names "")
foreach(_line IN LISTS _gmic_pins)
  if(_line MATCHES "^[ \t\r]*(#.*)?$")
    continue()
  endif()
  if(NOT _line MATCHES "^([A-Za-z0-9_.]+) *= *[^$#]*$")
    message(FATAL_ERROR "${GMIC_SOURCES_FILE}: not a \"NAME = value\" line from the first column "
      "(no variables or end-of-line comments): ${_line}")
  endif()
  if(CMAKE_MATCH_1 IN_LIST _names)
    message(FATAL_ERROR "${GMIC_SOURCES_FILE}: ${CMAKE_MATCH_1} is assigned more than once")
  endif()
  list(APPEND _names "${CMAKE_MATCH_1}")
endforeach()
function(gmic_sources_var _name _out)
  set(_value "")
  set(_count 0)
  foreach(_line IN LISTS _gmic_pins)
    if(_line MATCHES "^([A-Za-z0-9_.]+) *= *(.*)$" AND CMAKE_MATCH_1 STREQUAL _name)
      string(STRIP "${CMAKE_MATCH_2}" _value)
      math(EXPR _count "${_count} + 1")
    endif()
  endforeach()
  if(_count GREATER 1 OR _value MATCHES "\\\\$")
    message(FATAL_ERROR "${GMIC_SOURCES_FILE}: ${_name} must be assigned once, on one line")
  endif()
  set(${_out} "${_value}" PARENT_SCOPE)
endfunction()

gmic_sources_var(GMICVER _pin_gmicver)
if(NOT _pin_gmicver)
  message(FATAL_ERROR "Could not read GMICVER from ${GMIC_SOURCES_FILE}")
endif()
if(NOT GMICVER)
  set(GMICVER "${_pin_gmicver}")
elseif(NOT GMICVER STREQUAL _pin_gmicver)
  message(WARNING "GMICVER=${GMICVER} (command line or cache) overrides gmic_sources.txt's ${_pin_gmicver}: "
    "remove it with -UGMICVER")
endif()
gmic_sources_var("GMICSRC_SHA256_${GMICVER}" _pin_sha256)
if(NOT GMICSRC_SHA256)
  set(GMICSRC_SHA256 "${_pin_sha256}")
elseif(_pin_sha256 AND NOT GMICSRC_SHA256 STREQUAL _pin_sha256)
  message(WARNING "GMICSRC_SHA256=${GMICSRC_SHA256} (command line or cache) overrides gmic_sources.txt's "
    "GMICSRC_SHA256_${GMICVER}: remove it with -UGMICSRC_SHA256")
endif()
string(LENGTH "${GMICSRC_SHA256}" _len)
if(NOT _len EQUAL 64 OR NOT GMICSRC_SHA256 MATCHES "^[0-9a-fA-F]+$")
  message(FATAL_ERROR "No valid sha256 (64 hex digits) for G'MIC ${GMICVER}: add GMICSRC_SHA256_${GMICVER} "
    "to gmic_sources.txt (or configure with -DGMICSRC_SHA256=<sha256>)")
endif()
string(TOLOWER "${GMICSRC_SHA256}" GMICSRC_SHA256)
# All G'MIC sources come from the release tarball: since G'MIC 3, it includes the C API
# (gmic_libc.*) and the standard library with the community filters (gmic_stdlib_community.h,
# below). Only gmic_libc.h is used from the C API: gmic_plugin_call.cpp replaces gmic_libc.cpp.
set(GMICSOURCES CImg.h gmic.cpp gmic.h gmic_libc.h)
set(GMICSRC "gmic_${GMICVER}.tar.gz")
set(GMICSRCDIR "gmic-${GMICVER}")
# GreycLab/gmic on GitHub is the main G'MIC repository, and its releases (3.3.1 and later) attach
# gmic_<ver>.tar.gz: tried first, then gmic.eu (which may answer 403, seen in 2026) and the MacPorts
# mirror.
set(GMICSRC_URLS
  "https://github.com/GreycLab/gmic/releases/download/v.${GMICVER}/${GMICSRC}"
  "https://gmic.eu/files/source/${GMICSRC}"
  "https://distfiles.macports.org/gmic/${GMICSRC}")

# The tarball: GMIC_TARBALL if given (downloaded there if missing), else a gmic_<version>.tar.gz in
# the source directory (e.g. restored there by CI's cache), else downloaded into GMIC_DOWNLOAD_DIR.
set(GMIC_TARBALL "" CACHE FILEPATH "G'MIC source tarball (gmic_<version>.tar.gz), downloaded there if missing")
set(GMIC_DOWNLOAD_DIR "${CMAKE_BINARY_DIR}" CACHE PATH "Where to download the G'MIC source tarball")
if(GMIC_TARBALL)
  set(_gmic_tarball "${GMIC_TARBALL}")
  get_filename_component(_name "${GMIC_TARBALL}" NAME)
  if(NOT _name STREQUAL GMICSRC)
    message(WARNING "GMIC_TARBALL=${GMIC_TARBALL} (command line or cache) is not ${GMICSRC}: "
      "its checksum is verified anyway; remove it with -UGMIC_TARBALL")
  endif()
elseif(EXISTS "${PROJECT_SOURCE_DIR}/${GMICSRC}")
  set(_gmic_tarball "${PROJECT_SOURCE_DIR}/${GMICSRC}")
else()
  set(_gmic_tarball "${GMIC_DOWNLOAD_DIR}/${GMICSRC}")
endif()

if(EXISTS "${_gmic_tarball}")
  file(SHA256 "${_gmic_tarball}" _sum)
  if(NOT _sum STREQUAL GMICSRC_SHA256)
    message(FATAL_ERROR "${_gmic_tarball}: checksum mismatch (got ${_sum}, expected ${GMICSRC_SHA256}): remove it")
  endif()
else()
  # Each mirror in turn (gmic.eu may answer 403). The checksum is verified here,
  # not with file(DOWNLOAD EXPECTED_HASH), which stops at the first mismatch instead of trying
  # the next mirror; the file gets its final name only once verified.
  set(_tmp "${_gmic_tarball}.tmp")
  set(_ok FALSE)
  foreach(_url IN LISTS GMICSRC_URLS)
    # up to 3 tries, 5 s apart, except after an HTTP 4xx answer (e.g. gmic.eu's 403)
    foreach(_try 1 2 3)
      if(_try GREATER 1)
        execute_process(COMMAND "${CMAKE_COMMAND}" -E sleep 5)
      endif()
      message(STATUS "Fetching ${_url}")
      file(DOWNLOAD "${_url}" "${_tmp}" STATUS _status LOG _log TLS_VERIFY ON INACTIVITY_TIMEOUT 60)
      list(GET _status 0 _code)
      if(_code EQUAL 0)
        break()
      endif()
      list(GET _status 1 _msg)
      message(WARNING "Could not download ${_url}: ${_msg}")
      if(_log MATCHES "HTTP/[0-9.]+ 4[0-9][0-9]")
        break()
      endif()
    endforeach()
    if(_code EQUAL 0)
      file(SHA256 "${_tmp}" _sum)
      if(_sum STREQUAL GMICSRC_SHA256)
        file(RENAME "${_tmp}" "${_gmic_tarball}")
        set(_ok TRUE)
        break()
      endif()
      message(WARNING "Checksum mismatch for ${_url} (got ${_sum})")
    endif()
  endforeach()
  file(REMOVE "${_tmp}")
  if(NOT _ok)
    string(REPLACE ";" "\n  " _urls "${GMICSRC_URLS}")
    message(FATAL_ERROR "Could not download a valid ${GMICSRC}: download it from one of\n  ${_urls}\n"
      "and configure with -DGMIC_TARBALL=<path>")
  endif()
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_gmic_tarball}")

# The G'MIC sources of the build (GMICSOURCES, and for GPL builds the standard library with the
# community filters, gmic_stdlib_community.h; COMMERCIAL builds generate it, below), extracted
# into GMIC_EXTRACT_DIR, patched there, then copied into GMIC_SRC_DIR, which is what the build
# uses. The copy only writes files that changed, so that a new extraction doesn't recompile
# anything. Patching takes tens of seconds (awk on CImg.h), so it is only redone when its inputs
# change: a stamp in GMIC_SRC_DIR keys the result on the tarball's sha256, the extracted files and
# each patch script (name, file and sha256).
set(GMIC_EXTRACT_DIR "${CMAKE_BINARY_DIR}/gmic-extract")
set(GMIC_SRC_DIR "${CMAKE_BINARY_DIR}/gmic")
set(_gmic_extract ${GMICSOURCES})
if(OFX_LICENSE STREQUAL "GPL")
  list(APPEND _gmic_extract gmic_stdlib_community.h)
endif()

# Local patches to the extracted G'MIC sources, read from gmic_sources.txt (see patches/), applied
# in its order: CIMG_PATCHES on CImg.h, GMIC_PATCHES on gmic.cpp. Each runs as
# "sh patches/<script> <file>" on the fresh copy, and fails if the source doesn't contain the
# expected code (e.g. after a GMICVER bump). The patches need a POSIX sh (Linux, macOS, MSYS2; with
# MSVC, e.g. Git for Windows' sh.exe in the PATH, or -DGMIC_PATCH_SH=<path>).
gmic_sources_var(CIMG_PATCHES _pin_cimg_patches)
if(NOT _pin_cimg_patches)
  message(FATAL_ERROR "Could not read CIMG_PATCHES from ${GMIC_SOURCES_FILE}")
endif()
gmic_sources_var(GMIC_PATCHES _pin_gmic_patches)
if(NOT _pin_gmic_patches)
  message(FATAL_ERROR "Could not read GMIC_PATCHES from ${GMIC_SOURCES_FILE}")
endif()
set(_patches "")
set(_gmic_patch_lists "_pin_cimg_patches:CImg.h" "_pin_gmic_patches:gmic.cpp")
find_program(GMIC_PATCH_SH sh)
set(_gmic_patch_key "${GMICVER} ${GMICSRC_SHA256} ${_gmic_extract} sh=${GMIC_PATCH_SH}")
foreach(_entry IN LISTS _gmic_patch_lists)
  string(REPLACE ":" ";" _entry "${_entry}")
  list(GET _entry 0 _var)
  list(GET _entry 1 _file)
  separate_arguments(_scripts UNIX_COMMAND "${${_var}}")
  foreach(_script IN LISTS _scripts)
    if(NOT _script MATCHES "^patches/[A-Za-z0-9_.-]+\\.sh$" OR NOT EXISTS "${PROJECT_SOURCE_DIR}/${_script}")
      message(FATAL_ERROR "${GMIC_SOURCES_FILE}: ${_var} lists '${_script}', not an existing patches/<name>.sh")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/${_script}")
    file(SHA256 "${PROJECT_SOURCE_DIR}/${_script}" _sum)
    list(APPEND _patches "${_script}:${_file}")
    string(APPEND _gmic_patch_key " ${_script}:${_file}:${_sum}")
  endforeach()
endforeach()

set(_stamp "${GMIC_SRC_DIR}/sources.stamp")
set(_old_key "")
if(EXISTS "${_stamp}")
  file(READ "${_stamp}" _old_key)
  foreach(_f IN LISTS _gmic_extract)
    if(NOT EXISTS "${GMIC_SRC_DIR}/${_f}")
      set(_old_key "")
    endif()
  endforeach()
endif()
if(NOT _old_key STREQUAL _gmic_patch_key)
  file(REMOVE "${_stamp}")
  file(REMOVE_RECURSE "${GMIC_EXTRACT_DIR}")
  set(_patterns "")
  foreach(_f IN LISTS _gmic_extract)
    list(APPEND _patterns "${GMICSRCDIR}/src/${_f}")
  endforeach()
  file(MAKE_DIRECTORY "${GMIC_EXTRACT_DIR}")
  execute_process(COMMAND "${CMAKE_COMMAND}" -E tar xzf "${_gmic_tarball}" ${_patterns}
    WORKING_DIRECTORY "${GMIC_EXTRACT_DIR}" RESULT_VARIABLE _r)
  if(NOT _r EQUAL 0)
    message(FATAL_ERROR "Could not extract the G'MIC sources from ${_gmic_tarball}")
  endif()
  set(_gmic_extracted "${GMIC_EXTRACT_DIR}/${GMICSRCDIR}/src")
  foreach(_f IN LISTS _gmic_extract)
    if(NOT EXISTS "${_gmic_extracted}/${_f}")
      message(FATAL_ERROR "${GMICSRC} has no ${GMICSRCDIR}/src/${_f}")
    endif()
  endforeach()
  foreach(_patch IN LISTS _patches)
    string(REPLACE ":" ";" _patch "${_patch}")
    list(GET _patch 0 _script)
    list(GET _patch 1 _file)
    if(NOT GMIC_PATCH_SH)
      message(FATAL_ERROR "sh not found: it is needed to apply ${_script} to ${_file}: put a POSIX sh "
        "in the PATH, or configure with -DGMIC_PATCH_SH=<path to sh>")
    endif()
    execute_process(COMMAND "${GMIC_PATCH_SH}" "${PROJECT_SOURCE_DIR}/${_script}" "${_file}"
      WORKING_DIRECTORY "${_gmic_extracted}" RESULT_VARIABLE _r)
    if(NOT _r EQUAL 0)
      message(FATAL_ERROR "${_script} failed on ${_file} of G'MIC ${GMICVER}")
    endif()
  endforeach()
  foreach(_f IN LISTS _gmic_extract)
    configure_file("${_gmic_extracted}/${_f}" "${GMIC_SRC_DIR}/${_f}" COPYONLY)
  endforeach()
  file(REMOVE_RECURSE "${GMIC_EXTRACT_DIR}")
  file(WRITE "${_stamp}" "${_gmic_patch_key}")
endif()

# pocketfft (FFT=pocketfft): pocketfft_hdronly.h, at gmic_sources.txt's POCKETFFTCOMMIT and checked
# against its POCKETFFT_SHA256, patched by gmic_sources.txt's POCKETFFT_PATCHES into GMIC_SRC_DIR
# (next to CImg.h, which includes it), and its licence (BSD-3-Clause, which asks for its notice in
# binary distributions), from the header's first comment, into GMIC_GENERATED_DIR.
set(GMIC_GENERATED_DIR "${CMAKE_BINARY_DIR}/generated")
if(GMIC_FFT STREQUAL "pocketfft")
  gmic_sources_var(POCKETFFTCOMMIT _pocketfft_commit)
  gmic_sources_var(POCKETFFT_SHA256 _pocketfft_sha256)
  if(NOT _pocketfft_commit OR NOT _pocketfft_sha256)
    message(FATAL_ERROR "Could not read POCKETFFTCOMMIT and POCKETFFT_SHA256 from ${GMIC_SOURCES_FILE}")
  endif()
  # The header: GMIC_POCKETFFT if given (downloaded there if missing, as GMIC_TARBALL), else one in
  # the source directory (e.g. restored there by CI's cache), else downloaded into
  # ${CMAKE_BINARY_DIR}/pocketfft (not an include directory: it isn't patched). Each is checked
  # against POCKETFFT_SHA256.
  set(GMIC_POCKETFFT "" CACHE FILEPATH "pocketfft_hdronly.h (at gmic_sources.txt's POCKETFFTCOMMIT), downloaded there if missing")
  set(_pocketfft "${GMIC_SRC_DIR}/pocketfft_hdronly.h")
  set(_pocketfft_download "${CMAKE_BINARY_DIR}/pocketfft/pocketfft_hdronly.h")
  if(GMIC_POCKETFFT)
    set(_pocketfft_src "${GMIC_POCKETFFT}")
  elseif(EXISTS "${PROJECT_SOURCE_DIR}/pocketfft_hdronly.h")
    set(_pocketfft_src "${PROJECT_SOURCE_DIR}/pocketfft_hdronly.h")
  else()
    set(_pocketfft_src "${_pocketfft_download}")
  endif()
  set(_sum "")
  if(EXISTS "${_pocketfft_src}")
    file(SHA256 "${_pocketfft_src}" _sum)
    if(NOT _sum STREQUAL _pocketfft_sha256 AND NOT _pocketfft_src STREQUAL _pocketfft_download)
      message(FATAL_ERROR "${_pocketfft_src}: checksum mismatch (got ${_sum}, expected ${_pocketfft_sha256}): remove it")
    endif()
  endif()
  if(NOT _sum STREQUAL _pocketfft_sha256)
    set(_url "https://raw.githubusercontent.com/mreineck/pocketfft/${_pocketfft_commit}/pocketfft_hdronly.h")
    message(STATUS "Fetching ${_url}")
    file(DOWNLOAD "${_url}" "${_pocketfft_src}.tmp" STATUS _status TLS_VERIFY ON INACTIVITY_TIMEOUT 60)
    list(GET _status 0 _code)
    if(_code EQUAL 0)
      file(SHA256 "${_pocketfft_src}.tmp" _sum)
    endif()
    if(NOT _code EQUAL 0 OR NOT _sum STREQUAL _pocketfft_sha256)
      file(REMOVE "${_pocketfft_src}.tmp")
      message(FATAL_ERROR "Could not download a valid ${_url} (${_status}, sha256 ${_sum}): download it "
        "and configure with -DGMIC_POCKETFFT=<path>")
    endif()
    file(RENAME "${_pocketfft_src}.tmp" "${_pocketfft_src}")
  endif()
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_pocketfft_src}")
  # The local patches (POCKETFFT_PATCHES, as CIMG_PATCHES above), applied to a copy at each
  # configure (it takes a fraction of a second); configure_file only writes GMIC_SRC_DIR's header
  # when the result changed.
  gmic_sources_var(POCKETFFT_PATCHES _pin_pocketfft_patches)
  if(NOT _pin_pocketfft_patches)
    message(FATAL_ERROR "Could not read POCKETFFT_PATCHES from ${GMIC_SOURCES_FILE}")
  endif()
  if(NOT GMIC_PATCH_SH)
    message(FATAL_ERROR "sh not found: it is needed to apply POCKETFFT_PATCHES to pocketfft_hdronly.h: put a "
      "POSIX sh in the PATH, or configure with -DGMIC_PATCH_SH=<path to sh>")
  endif()
  set(_pocketfft_patch_dir "${CMAKE_BINARY_DIR}/pocketfft/patched")
  file(REMOVE_RECURSE "${_pocketfft_patch_dir}")
  file(MAKE_DIRECTORY "${_pocketfft_patch_dir}")
  configure_file("${_pocketfft_src}" "${_pocketfft_patch_dir}/pocketfft_hdronly.h" COPYONLY)
  separate_arguments(_scripts UNIX_COMMAND "${_pin_pocketfft_patches}")
  foreach(_script IN LISTS _scripts)
    if(NOT _script MATCHES "^patches/[A-Za-z0-9_.-]+\\.sh$" OR NOT EXISTS "${PROJECT_SOURCE_DIR}/${_script}")
      message(FATAL_ERROR "${GMIC_SOURCES_FILE}: POCKETFFT_PATCHES lists '${_script}', not an existing patches/<name>.sh")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/${_script}")
    execute_process(COMMAND "${GMIC_PATCH_SH}" "${PROJECT_SOURCE_DIR}/${_script}" pocketfft_hdronly.h
      WORKING_DIRECTORY "${_pocketfft_patch_dir}" RESULT_VARIABLE _r)
    if(NOT _r EQUAL 0)
      message(FATAL_ERROR "${_script} failed on pocketfft_hdronly.h (POCKETFFTCOMMIT ${_pocketfft_commit})")
    endif()
  endforeach()
  configure_file("${_pocketfft_patch_dir}/pocketfft_hdronly.h" "${_pocketfft}" COPYONLY)
  file(REMOVE_RECURSE "${_pocketfft_patch_dir}")
  file(READ "${_pocketfft}" _text)
  string(FIND "${_text}" "\n*/\n" _end)
  string(SUBSTRING "${_text}" 0 3 _begin)
  if(NOT _begin STREQUAL "/*\n" OR _end LESS 0)
    message(FATAL_ERROR "${_pocketfft}: no licence comment at the top")
  endif()
  math(EXPR _len "${_end} - 2")
  string(SUBSTRING "${_text}" 3 ${_len} _licence) # the lines between "/*" and "*/"
  if(NOT _licence MATCHES "Redistributions in binary form must reproduce")
    message(FATAL_ERROR "${_pocketfft}: the licence comment is not the expected one")
  endif()
  # file(WRITE) writes CRLF line endings on Windows: configure_file(NEWLINE_STYLE UNIX) writes LF,
  # (the licence has no @VAR@ for @ONLY to replace). The input is kept (a configure dependency) and
  # is not <output>.tmp, through which configure_file writes.
  file(WRITE "${GMIC_GENERATED_DIR}/Licence_pocketfft.txt.in" "${_licence}")
  configure_file("${GMIC_GENERATED_DIR}/Licence_pocketfft.txt.in" "${GMIC_GENERATED_DIR}/Licence_pocketfft.txt"
    @ONLY NEWLINE_STYLE UNIX)
endif()

# COMMERCIAL: gmic_stdlib_community.h is generated by gmic_stdlib_commercial.sh. It builds a G'MIC
# interpreter from the tarball with this compiler and zlib (a few minutes the first time; kept in
# ${CMAKE_BINARY_DIR}/gmic_stdlib_commercial), and downloads the community files of
# gmic_stdlib_commercial.txt. It runs again when the tarball, the script or
# gmic_stdlib_commercial.txt change. It also writes gmic_stdlib_community.h.commands, the commands
# that the library defines, for the hot fixes (below).
if(OFX_LICENSE STREQUAL "COMMERCIAL")
  if(NOT GMIC_PATCH_SH)
    message(FATAL_ERROR "sh not found: it is needed to generate the COMMERCIAL standard library "
      "(gmic_stdlib_commercial.sh); configure with -DGMIC_PATCH_SH=<path to sh>")
  endif()

  pkg_check_modules(GMIC_ZLIB REQUIRED zlib)

  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_script}" "${_conf}")
  set(_out "${CMAKE_BINARY_DIR}/gmic_stdlib_commercial/gmic_stdlib_community.h")
  file(SHA256 "${_script}" _k1)
  file(SHA256 "${_conf}" _k2)
  set(_key "${GMICVER} ${GMICSRC_SHA256} ${_k1} ${_k2} ${CMAKE_CXX_COMPILER}")
  set(_stamp "${CMAKE_BINARY_DIR}/gmic_stdlib_commercial/stamp")
  set(_old_key "")
  if(EXISTS "${_stamp}" AND EXISTS "${_out}" AND EXISTS "${_out}.commands")
    file(READ "${_stamp}" _old_key)
  endif()
  if(NOT _old_key STREQUAL _key)
    file(REMOVE "${_stamp}")

    string(REPLACE ";" " " _zlib_libs "${GMIC_ZLIB_LDFLAGS}")

        "${GMIC_PATCH_SH}" "${_script}" "${_gmic_tarball}" "${GMICVER}" "${_out}"
      WORKING_DIRECTORY "${CMAKE_BINARY_DIR}" RESULT_VARIABLE _r)
    if(NOT _r EQUAL 0)
      message(FATAL_ERROR "gmic_stdlib_commercial.sh failed (see above)")
    endif()
    file(WRITE "${_stamp}" "${_key}")
  endif()
  file(STRINGS "${_out}" _marker LIMIT_COUNT 10 REGEX "^ #  openfx-gmic : COMMERCIAL standard library")
  if(NOT _marker)
    message(FATAL_ERROR "${_out} is not the COMMERCIAL standard library")
  endif()
  configure_file("${_out}" "${GMIC_SRC_DIR}/gmic_stdlib_community.h" COPYONLY)
  set(GMIC_STDLIB_COMMANDS "${_out}.commands")
endif()

# The sources in the source directory include their headers with #include "...", which looks first
# in the source directory: a G'MIC header left there (e.g. by the former Makefile build, or copied
# by hand) would be used instead of the one in GMIC_SRC_DIR. That is harmless only if it is the same
# file. This is checked at configure time, again when such a header changes, but not when one
# appears after the configure (run cmake again then).
foreach(_f IN LISTS GMICSOURCES)
  if(_f MATCHES "\\.h$" AND EXISTS "${PROJECT_SOURCE_DIR}/${_f}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/${_f}")
    file(SHA256 "${PROJECT_SOURCE_DIR}/${_f}" _stray)
    file(SHA256 "${GMIC_SRC_DIR}/${_f}" _ours)
    if(NOT _stray STREQUAL _ours)
      message(FATAL_ERROR "${PROJECT_SOURCE_DIR}/${_f} is not the ${_f} of this build (G'MIC ${GMICVER}, "
        "${OFX_LICENSE}, FFT=${GMIC_FFT}; left by an old Makefile build?), "
        "and would be included instead of ${GMIC_SRC_DIR}/${_f}: remove CImg.h, gmic.cpp, gmic.h, "
        "gmic_libc.h and gmic_stdlib_community.h from the source directory")
    endif()
  endif()
endforeach()

# The libraries, through pkg-config (so that the intended ones are found, e.g. MacPorts' zlib rather
# than the macOS SDK's): zlib, libpng, plus fftw3 and fftw3_threads with FFT=fftw3. No libcurl, on
# any system: G'MIC downloads filter data with the system's curl command (cimg::load_network;
# gmic_plugin_call.cpp sets its path: macOS /usr/bin/curl, Windows 10 1803 and later
# System32\curl.exe, other systems curl or wget from PATH).
#
# Linux: GMIC.ofx needs only system libraries (libc, libm, libstdc++,
# libgcc_s, libgomp), so that one binary runs on every distribution
# (Natron 2.6's baseline: Rocky Linux 9, glibc 2.34, GCC 11). zlib, libpng and FFTW are linked
# statically with GMIC_STATIC_DEPS (ON by default on Linux): their static archives (libz.a,
# libpng16.a, libfftw3.a, libfftw3_threads.a, built with -fPIC, which distributions' archives may
# not be: see .github/workflows/build_static_deps.sh) must be in pkg-config's libdir or the
# linker's search path. -DGMIC_STATIC_DEPS=OFF links the shared libraries, e.g. for
# distribution packages. The version script of add_ofx_plugin keeps their symbols local.
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  option(GMIC_STATIC_DEPS "Linux: link zlib, libpng and FFTW statically (GMIC.ofx needs only system libraries)" ON)
else()
  set(GMIC_STATIC_DEPS OFF)
endif()
# MSVC: zlib and libpng through CMake's FindZLIB and FindPNG, with no pkg-config: e.g. vcpkg's
# x64-windows-static triplet (static libraries built with /MT), with the vcpkg.json manifest and
# vcpkg's toolchain file (see README). FFT=fftw3 still uses pkg-config.
if(MSVC AND NOT GMIC_FFT STREQUAL "fftw3")
  find_package(ZLIB REQUIRED)
  find_package(PNG REQUIRED)
  target_link_libraries(GMIC PRIVATE PNG::PNG ZLIB::ZLIB)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GMIC_DEPS REQUIRED IMPORTED_TARGET libpng zlib ${_fft_module})
  if(GMIC_FFT STREQUAL "fftw3")
    set(_fftw3_threads fftw3_threads)
  else()
    set(_fftw3_threads "")
  endif()
  if(GMIC_STATIC_DEPS)
    target_include_directories(GMIC PRIVATE ${GMIC_DEPS_STATIC_INCLUDE_DIRS})
    target_compile_options(GMIC PRIVATE ${GMIC_DEPS_STATIC_CFLAGS_OTHER})
    # the libdirs of libpng and zlib first, then FFTW's, then the -L directories: the archives of
    # build_static_deps.sh come before those of FFTW's libdir (e.g. /usr/lib/<arch>, which may also
    # have non-PIC libpng and zlib archives)
    set(_dirs "")
    foreach(_module libpng zlib ${_fft_module})
      string(REGEX REPLACE "[<>=].*" "" _module "${_module}")
      pkg_get_variable(_libdir ${_module} libdir)
      list(APPEND _dirs ${_libdir})
    endforeach()
    list(APPEND _dirs ${GMIC_DEPS_STATIC_LIBRARY_DIRS})
    foreach(_lib ${_fftw3_threads} ${GMIC_DEPS_STATIC_LIBRARIES})
      if(_lib MATCHES "^(m|dl|pthread|rt|c)$") # system libraries
        target_link_libraries(GMIC PRIVATE ${_lib})
        continue()
      endif()
      # cached (so that -DGMIC_STATIC_<lib>=<archive> can choose another one): after changing
      # PKG_CONFIG_PATH or the archives' location, configure a new build directory, or with
      # -U 'GMIC_STATIC_*'
      find_library(GMIC_STATIC_${_lib} NAMES lib${_lib}.a HINTS ${_dirs})
      if(NOT GMIC_STATIC_${_lib})
        message(FATAL_ERROR "GMIC_STATIC_DEPS: lib${_lib}.a not found (in ${_dirs} or the default paths): "
          "build them with -fPIC (.github/workflows/build_static_deps.sh <prefix>, then "
          "PKG_CONFIG_PATH=<prefix>/lib/pkgconfig), or configure with -DGMIC_STATIC_DEPS=OFF")
      endif()
      target_link_libraries(GMIC PRIVATE ${GMIC_STATIC_${_lib}})
    endforeach()
  else()
    target_link_libraries(GMIC PRIVATE PkgConfig::GMIC_DEPS)
    if(_fftw3_threads)
      pkg_get_variable(_fftw3_libdir fftw3 libdir)
      find_library(FFTW3_THREADS_LIBRARY fftw3_threads HINTS ${_fftw3_libdir} ${GMIC_DEPS_LIBRARY_DIRS})
      if(NOT FFTW3_THREADS_LIBRARY)
        message(FATAL_ERROR "fftw3_threads not found (in ${_fftw3_libdir})")
      endif()
      target_link_libraries(GMIC PRIVATE ${FFTW3_THREADS_LIBRARY})
    endif()
  endif()
endif()

# OpenMP (OFX_USE_OPENMP). MSVC: /openmp (OpenMP 2.0, which CImg supports: it drops collapse()
# with MSVC), whose runtime, vcomp140.dll, is copied next to GMIC.ofx (below). (CMake's FindOpenMP
# is not used with MSVC: it may pick /openmp:llvm or /openmp:experimental, and its check failed with
# the static C runtime. /openmp:llvm is not usable: its libomp140 DLL isn't redistributable, and it
# aborts in CImg's nested parallel regions.)
if(OFX_USE_OPENMP AND MSVC)
  target_compile_options(GMIC PRIVATE /openmp)
  target_compile_definitions(GMIC PRIVATE cimg_use_openmp)
  # Debug builds (/MTd, /MDd) link the release OpenMP runtime too, vcomp140.dll (not
  # vcomp140d.dll, which is not redistributable), so that the bundle and its manifest are the same
  # in every configuration. (vcomp's API is C: it doesn't depend on the C runtime of GMIC.ofx.)
  target_link_options(GMIC PRIVATE /NODEFAULTLIB:vcompd.lib)
  target_link_libraries(GMIC PRIVATE vcomp.lib)
elseif(OFX_USE_OPENMP)
  ofx_target_openmp(GMIC REQUIRED)
  target_compile_definitions(GMIC PRIVATE cimg_use_openmp)
endif()

# MSVC with OpenMP: vcomp140.dll, from the redistributable files of this compiler
# (VC/Redist/MSVC/<version>/<arch>/Microsoft.VC<nnn>.OpenMP/vcomp140.dll: not the debug_nonredist,
# onecore or OpenMP.LLVM copies), is copied app-local into Contents/<ARCHDIR> next to GMIC.ofx, in
# the build tree and at install. Hosts load the plug-in with LoadLibrary(<path>), which doesn't
# look for its dependencies in its directory: a manifest embedded into GMIC.ofx lists vcomp140.dll
# as a file of the plug-in's private assembly, so that the loader takes the bundled copy (as the
# MSYS2 artifacts' bundle_libs.sh does). -DGMIC_VCOMP_DLL=<path> chooses another copy.
if(OFX_USE_OPENMP AND MSVC)
  # the Redist folders to look in, and the manifest's processorArchitecture. Arm64EC (and Arm64X)
  # code uses the arm64 folder's vcomp140.dll, an Arm64X image that loads in Arm64EC and in ARM64
  # processes, unless an arm64ec folder exists. (An Arm64X GMIC.ofx gets the Arm64EC build's amd64
  # manifest, which ARM64 processes also accept.)
  string(TOLOWER "${CMAKE_CXX_COMPILER_ARCHITECTURE_ID}" _vcomp_arch)
  # (CMake versions that don't report ARM64EC as the architecture: -arm64EC or /arm64EC in the flags)
  string(TOLOWER "${CMAKE_CXX_FLAGS}" _flags)
  if(_flags MATCHES "(^| )[-/]arm64ec( |$)")
    set(_vcomp_arch arm64ec)
  endif()
  set(_vcomp_arches "${_vcomp_arch}")
  if(_vcomp_arch STREQUAL "x64")
    set(_manifest_arch amd64)
  elseif(_vcomp_arch STREQUAL "arm64")
    set(_manifest_arch arm64)
  elseif(_vcomp_arch STREQUAL "arm64ec")
    set(_vcomp_arches arm64ec arm64)
    set(_manifest_arch amd64)
  elseif(_vcomp_arch STREQUAL "x86")
    set(_manifest_arch x86)
  else()
    message(FATAL_ERROR "OpenMP with MSVC: no vcomp140.dll for the architecture '${CMAKE_CXX_COMPILER_ARCHITECTURE_ID}' "
      "(configure with -DOFX_USE_OPENMP=OFF)")
  endif()
  if(NOT GMIC_VCOMP_DLL)
    # VCToolsRedistDir (set by vcvars), else the Redist directory next to the compiler's Tools
    # directory (<VC>/Tools/MSVC/<version>/bin/Host<arch>/<arch>/cl.exe)
    set(_redist_dirs "")
    if(DEFINED ENV{VCToolsRedistDir})
      file(TO_CMAKE_PATH "$ENV{VCToolsRedistDir}" _dir)
      list(APPEND _redist_dirs "${_dir}")
    endif()
    get_filename_component(_vc "${CMAKE_CXX_COMPILER}" DIRECTORY)
    get_filename_component(_vc "${_vc}/../../../../../.." ABSOLUTE)
    file(GLOB _dirs LIST_DIRECTORIES true "${_vc}/Redist/MSVC/*")
    list(SORT _dirs COMPARE NATURAL ORDER DESCENDING)
    list(APPEND _redist_dirs ${_dirs})
    foreach(_dir IN LISTS _redist_dirs)
      foreach(_a IN LISTS _vcomp_arches)
        file(GLOB _dll "${_dir}/${_a}/Microsoft.VC*.OpenMP/vcomp140.dll")
        if(_dll)
          break()
        endif()
      endforeach()
      if(_dll)
        list(GET _dll 0 _dll)
        set(GMIC_VCOMP_DLL "${_dll}" CACHE FILEPATH "MSVC's OpenMP runtime, copied next to GMIC.ofx")
        break()
      endif()
    endforeach()
  endif()
  if(NOT GMIC_VCOMP_DLL OR NOT EXISTS "${GMIC_VCOMP_DLL}")
    message(FATAL_ERROR "OpenMP with MSVC: vcomp140.dll not found in the redistributable files of the compiler "
      "(VC/Redist/MSVC/<version>/${_vcomp_arch}/Microsoft.VC<nnn>.OpenMP): configure with "
      "-DGMIC_VCOMP_DLL=<path>, or -DOFX_USE_OPENMP=OFF")
  endif()
  string(TOLOWER "${GMIC_VCOMP_DLL}" _dll) # (VC145's arm64 folder is Microsoft.VC145.OPENMP)
  if(_dll MATCHES "debug_nonredist|/onecore/|openmp\\.llvm|debugopenmp|vcomp140d")
    message(FATAL_ERROR "GMIC_VCOMP_DLL=${GMIC_VCOMP_DLL} is not the redistributable vcomp140.dll")
  endif()
  message(STATUS "OpenMP (MSVC /openmp): ${GMIC_VCOMP_DLL} is copied next to GMIC.ofx (manifest: ${_manifest_arch})")
  # its notice: vcomp140.dll is Microsoft's, not under the plug-in's licence
  ofx_plugin_resources(GMIC GMIC_OFX/vcomp140-NOTICE.txt)
  configure_file(GMIC_OFX/GMIC.ofx.manifest.in "${GMIC_GENERATED_DIR}/GMIC.ofx.manifest" @ONLY)
  target_sources(GMIC PRIVATE "${GMIC_GENERATED_DIR}/GMIC.ofx.manifest")
  add_custom_command(TARGET GMIC POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${GMIC_VCOMP_DLL}" "$<TARGET_FILE_DIR:GMIC>/vcomp140.dll"
    VERBATIM)
  install(FILES "${GMIC_VCOMP_DLL}" RENAME vcomp140.dll
    DESTINATION "${PLUGIN_INSTALLDIR}/GMIC.ofx.bundle/Contents/${ARCHDIR}")
endif()

# Arm64X (MSVC, two builds, Microsoft's recipe: https://learn.microsoft.com/windows/arm/arm64x-build):
# GMIC.ofx holds both ARM64 and Arm64EC code, and loads in native ARM64 hosts and in Arm64EC hosts
# (and x64 hosts under emulation).
# - GMIC_ARM64X=ARM64: an ARM64 build, which also writes its linker inputs (objects, libraries) with
#   their full paths into GMIC_ARM64X_RSP (MSVC 17.11 or later: /LINKREPROFULLPATHRSP);
# - GMIC_ARM64X=ARM64EC: an Arm64EC build (/arm64EC), configured after the ARM64 build, which links
#   GMIC.ofx with those inputs as an Arm64X image (/machine:arm64x). It installs into Win-arm64ec.
set(GMIC_ARM64X "" CACHE STRING "MSVC Arm64X: ARM64 (first build) or ARM64EC (second build); empty: no Arm64X")
set(GMIC_ARM64X_RSP "" CACHE FILEPATH "Arm64X: the linker inputs of the ARM64 build (written by it, read by the Arm64EC build)")
if(GMIC_ARM64X)
  if(NOT MSVC OR NOT GMIC_ARM64X MATCHES "^(ARM64|ARM64EC)$" OR NOT GMIC_ARM64X_RSP)
    message(FATAL_ERROR "GMIC_ARM64X=ARM64 or ARM64EC needs MSVC and GMIC_ARM64X_RSP=<file>")
  endif()
  string(TOUPPER "${CMAKE_CXX_COMPILER_ARCHITECTURE_ID}" _arch)
  if(NOT _arch STREQUAL GMIC_ARM64X)
    message(FATAL_ERROR "GMIC_ARM64X=${GMIC_ARM64X}, but the compiler targets ${CMAKE_CXX_COMPILER_ARCHITECTURE_ID}")
  endif()
  if(GMIC_ARM64X STREQUAL "ARM64")
    target_link_options(GMIC PRIVATE "/LINKREPROFULLPATHRSP:${GMIC_ARM64X_RSP}")
  else()
    if(NOT EXISTS "${GMIC_ARM64X_RSP}")
      message(FATAL_ERROR "${GMIC_ARM64X_RSP} not found: build the GMIC_ARM64X=ARM64 configuration first")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${GMIC_ARM64X_RSP}")
    # the linker writes it in UTF-16LE with CRLF line endings, one quoted input per line (also
    # /manifestinput:..., left out: the Arm64EC build embeds the same manifest)
    file(STRINGS "${GMIC_ARM64X_RSP}" _lines ENCODING UTF-16LE)
    set(_objs "")
    set(_libs "")
    foreach(_l IN LISTS _lines)
      string(STRIP "${_l}" _l)
      if(_l MATCHES "^\"(.*\\.[oO][bB][jJ])\"$")
        string(REPLACE "\\" "/" _f "${CMAKE_MATCH_1}")
        list(APPEND _objs "${_f}")
      elseif(_l MATCHES "^\"(.*\\.[lL][iI][bB])\"$")
        string(REPLACE "\\" "/" _f "${CMAKE_MATCH_1}")
        list(APPEND _libs "${_f}")
      endif()
    endforeach()
    if(NOT _objs)
      message(FATAL_ERROR "${GMIC_ARM64X_RSP}: no object files found")
    endif()
    list(LENGTH _objs _n_objs)
    list(LENGTH _libs _n_libs)
    message(STATUS "Arm64X: GMIC.ofx links the ${_n_objs} objects and ${_n_libs} libraries of the ARM64 build too")
    target_sources(GMIC PRIVATE ${_objs})
    target_link_options(GMIC PRIVATE /machine:arm64x ${_libs})
  endif()
endif()
